cmake_minimum_required(VERSION 3.16)
project(airport_bst_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(airport_core src/Airport.cpp src/AirportBST.cpp src/DataIO.cpp)
target_include_directories(airport_core PUBLIC src)
add_executable(airports src/main.cpp src/Experiment.cpp)
target_link_libraries(airports PRIVATE airport_core)
add_executable(airport_tests tests/test_airport_bst.cpp)
target_link_libraries(airport_tests PRIVATE airport_core)
enable_testing()
add_test(NAME airport_bst_tests COMMAND airport_tests)
set_tests_properties(airport_bst_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
